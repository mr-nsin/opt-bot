Python Implementation: Adaptive Kaufman Efficiency Ratio ATR

import numpy as np
import pandas as pd

def calculate_kaufman_adaptive_atr(df: pd.DataFrame, period: int = 10, fast_p: int = 2, slow_p: int = 30):
    """Calculates Kaufman Efficiency Ratio (ER) and scales ATR multiplier dynamically."""
    change = (df['close'] - df['close'].shift(period)).abs()
    volatility = (df['close'] - df['close'].shift(1)).abs().rolling(window=period).sum()
    er = np.where(volatility != 0, change / volatility, 0.0)
    
    # Scale ATR Multiplier between 2.2x (High Trend Efficiency) and 3.6x (Choppy Market Noise)
    adaptive_mult = 3.6 - (er * 1.4)
    return er, adaptive_mult
