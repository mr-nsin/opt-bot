// Rust Shared Memory Producer (mmap)
use memmap2::MmapMut;
use std::fs::OpenOptions;

pub struct SharedTickRing {
    mmap: MmapMut,
}
impl SharedTickRing {
    pub fn new(path: &str, size: usize) -> Self {
        let file = OpenOptions::new().read(true).write(true).create(true).open(path).unwrap();
        file.set_len(size as u64).unwrap();
        let mmap = unsafe { MmapMut::map_mut(&file).unwrap() };
        Self { mmap }
    }
}

❌ Top 10 Things NOT to Do (Latency & Stability Antipatterns)

  1. DO NOT use JSON.stringify / JSON.parse over stdio for High-Frequency Ticks: Causes high CPU spikes and latency delays (> 10ms).
  2. DO NOT perform row-by-row DataFrame iteration (iterrows): 100x slower than vectorized Polars or Numba operations.
  3. DO NOT call React setState on raw unthrottled WebSocket/IPC tick streams: Freezes V8 JavaScript main thread.
  4. DO NOT use Python's standard multiprocessing.Queue for HFT: Deep copy pickling overhead destroys throughput.
  5. DO NOT leave SQLite in DEFAULT journal mode: Causes write-lock thread blocking during trade execution.
  6. DO NOT re-render SVG chart components for real-time tick updates: HTML SVG DOM updates consume heavy CPU.
  7. DO NOT build single-file PyInstaller binaries (--onefile) for desktop apps: Incurs 2-second startup decompression penalty.
  8. DO NOT allocate new memory buffers inside tick processing loops: Triggers garbage collection pauses during live trading.
  9. DO NOT hold Python GIL locks during heavy matrix compute: Blocks concurrent background threads.
  10. DO NOT use un-indexed database queries for trade blotter history: Full table scans degrade UI response times.
